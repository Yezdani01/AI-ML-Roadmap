TO CREATE A TABLE 
(serial is generated automatically)

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY, 
    username VARCHAR(50),
    age INT,
    is_active BOOLEAN
);


INSERT INTO users (username, age, is_active) VALUES
('hamza', 24, true),
('safaa', 31, false),
('alex', 19, true);

-----> To display the values inside the TABLE
SELECT * FROM users;


-----> to display the specific data 
SELECT username, age FROM users;

-----> with any specific condition 
SELECT username, age FROM users WHERE age > 20;
SELECT * FROM users WHERE is_active = true AND age < 25;


------- FEW COMMANDS 

• \dt — Lists all tables in your current database.
• \d users — Shows the columns, data types, and primary key of the users table.
• \q — Quits the psql tool and returns you to your normal Mac terminal.

USING THE AND OR NOT 

SELECT username
FROM users 
WHERE (is_active = 'False') 
  AND NOT age  < 20;

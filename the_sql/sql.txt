TO CREATE A TABLE 
(serial is generated automatically)

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY, 
    username VARCHAR(50),
    age INT,
    is_active BOOLEAN
);

-----> To display the values inside the TABLE
SELECT * FROM users;


-----> to display the specific data 
SELECT username, age FROM users;

-----> with any specific condition 
SELECT username, age FROM users WHERE age > 20;
SELECT * FROM users WHERE is_active = true AND age < 25;

using IN 

SELECT * FROM users
WHERE age IN (19, 31);



BETWEEN operator


SELECT * FROM users
WHERE age BETWEEN 20 AND 32;




LIKE operator


for starting with 

select * from users
where username like 'h%'; 

for ending with 

select * from users
where username like '%a'; 


exactly 4 characters

select username from users WHERE username LIKE '____';


PostgreSQL also has ILIKE, which performs case-insensitive pattern matching.
This matches hamza despite the difference in capitalization.

SELECT * FROM users
WHERE username ILIKE 'HAMZA';

Concept	SQL syntax

OR	WHERE age > 25 OR is_active = TRUE
NOT	WHERE NOT is_active
IN	WHERE age IN (19, 24)
BETWEEN	WHERE age BETWEEN 20 AND 30
LIKE	WHERE username LIKE 'h%'
NULL	WHERE age IS NULL
NOT NULL	WHERE age IS NOT NULL
ORDER BY	ORDER BY age ASC

CREATE TABLE orders (
    order_id SERIAL PRIMARY KEY,
    product VARCHAR(100),
    user_id INT REFERENCES users(user_id)
);



INNER JOIN



SELECT users.username, orders.product
FROM users
INNER JOIN orders
ON users.user_id = orders.user_id;

LEFT JOIN


SELECT users.username, orders.product
FROM users
LEFT JOIN orders
ON users.user_id = orders.user_id;


RIGHT JOIN

SELECT users.username, orders.product
FROM users
RIGHT JOIN orders
ON users.user_id = orders.user_id;



insted of using the full name you can also use aliases
FROM users u
INNER JOIN orders o


SELECT u.username, o.product
FROM users AS u
INNER JOIN orders AS o
ON u.user_id = o.user_id;
